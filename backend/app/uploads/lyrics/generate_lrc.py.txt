from aeneas.executetask import ExecuteTask
from aeneas.task import Task

# 1) Configure la tâche
config_string = "task_language=fra|os_task_file_format=lrc"
task = Task(config_string=config_string,
            text_file_path="lyrics.txt",
            audio_file_path="jetemmene_au_vent.mp3",
            output_file_path="jetemmene_au_vent.lrc")

# 2) Lance le traitement
ExecuteTask(task).execute()

# 3) Sauvegarde le résultat
task.output_sync_map_file()
print("Fichier LRC généré :", task.output_file_path)
